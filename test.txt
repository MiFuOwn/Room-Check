import streamlit as st
import requests
import time

# กำหนด URL ของ Realtime Database
url = 'https://roomcheek-default-rtdb.asia-southeast1.firebasedatabase.app/data.json'

# ฟังก์ชันสำหรับการอัปเดตข้อมูลแบบ Real-time
def update_data(url):
    while True:
        # ส่งคำร้องขอ GET เพื่อรับข้อมูลจาก Realtime Database
        response = requests.get(url)

        # ตรวจสอบว่าคำร้องขอสำเร็จหรือไม่
        if response.status_code == 200:
            # แปลงข้อมูล JSON ให้เป็น Python dictionary
            data = response.json()

            # ตรวจสอบว่ามีข้อมูล Light และ Sound ใน dictionary หรือไม่
            if 'Light' in data and 'Sound' in data:
                light_value = data['Light']
                sound_value = data['Sound']
                print("Light:", light_value)
                print("Sound:", sound_value)
            else:
                print("ไม่พบข้อมูล Light หรือ Sound")
        else:
            print("เกิดข้อผิดพลาดในการรับข้อมูล:", response.status_code)
                    
        # หน่วงเวลาเพื่อให้เวลาอัปเดตข้อมูล
        time.sleep(1)
# เริ่มต้นการอัปเดตข้อมูลแบบ Real-time
update_data(url)
